- Create initramfs folder
    cd ~/qemu-lab

- Copy busy box to initramfs
    cp /bin/busybox initramfs/bin/

- Create the guest init script 
    cat > initramfs/init << 'EOF'
    #!/bin/busybox sh

    /bin/busybox mount -t proc none /proc
    /bin/busybox mount -t sysfs none /sys

    echo "=== telemetry guest booted ==="

    seq=1

    while [ $seq -le 5 ]
    do
    echo "{\"sensor\":\"temp\",\"value\":72.$seq,\"unit\":\"F\",\"seq\":$seq}"
    seq=$((seq + 1))
    /bin/busybox sleep 1
    done

    echo "=== telemetry complete ==="

    /bin/busybox poweroff -f
    EOF

    chmod +x initramfs/init

    **At this point BusyBox and /init inside it are in initramfa

→ package it as telemetry-initramfs.cpio.gz
    cd ~/qemu-lab/initramfs
    find . | cpio -H newc -o | gzip > ../telemetry-initramfs.cpio.gz
    cd ..
- Find your Ubuntu kernel
    ls /boot/vmlinuz-*

- Create a local readable copy of Linux kernel
    sudo cp /boot/vmlinuz-6.17.0-23-generic ~/qemu-lab/kernel/
    sudo chown $USER:$USER ~/qemu-lab/kernel/vmlinuz-6.17.0-23-generic

→ QEMU loads kernel + initramfs
    qemu-system-x86_64 \
    -machine q35 \
    -m 512 \
    -kernel kernel/vmlinuz-6.17.0-23-generic \
    -initrd telemetry-initramfs.cpio.gz \
    -append "console=ttyS0 rdinit=/init panic=-1" \
    -nographic \
    -serial file:logs/telemetry.log

    console=ttyS0 rdinit=/init panic=-1
        console=ttyS0 → serial console
        rdinit=/init → first process
        panic=-1 → don't auto reboot on panic
    serial capture
        serial file:logs/telemetry.log
        other valid entries stdout, socket, pipe, PTY, etc.

- QEMU is acting like boot ROM
    load kernel image
    load initramfs
    jump to kernel entry point
    kernel boots
    kernel executes /init
    /init emits telemetry
    serial redirected to host log

- BusyBox acted as
    shell
    mount utility
    sleep
    poweroff

- Linux kernel boots
    → mounts initramfs as root filesystem
        → Linux kernel unpacks initramfs into RAM    → executes /init
    → kernel runs /init
        → /init emits telemetry
    → QEMU serial redirects output to host log

- whole boot flow
    QEMU starts
    → loads kernel
    → loads initramfs
    → kernel initializes
    → kernel runs /init
    → /init emits telemetry
    → serial redirected to host file
    → guest powers off

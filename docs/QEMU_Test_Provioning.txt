- Create lab workspace
    mkdir -p ~/qemu-lab/{images,firmware,kernel}
    cd ~/qemu-lab

- Copy writeable firmware vars
    cp /usr/share/OVMF/OVMF_VARS_4M.fd firmware/  

- Directory Layout
    ~/qemu-lab/
    images/
    firmware/
        OVMF_VARS_4M.fd
    kernel/

- Install x86_64 QEMU
    sudo apt install qemu-system-x86
    which qemu-system-x86_64
    qemu-system-x86_64 --version    

- Smoke test
    qemu-system-x86_64 \
    -machine q35 \
    -m 1024 \
    -bios /usr/share/OVMF/OVMF_CODE_4M.fd \
    -drive if=pflash,format=raw,file=firmware/OVMF_VARS_4M.fd

- Add folders for logging and scripts
    cd ~/qemu-lab
    mkdir -p telemetry logs scripts

- First host-side telemetry producer
    cat > telemetry/canned_telemetry.jsonl << 'EOF'
    {"sensor":"temp","value":72.4,"unit":"F","seq":1}
    {"sensor":"rpm","value":1810,"unit":"rpm","seq":2}
    {"sensor":"voltage","value":12.1,"unit":"V","seq":3}
    {"sensor":"pressure","value":31.8,"unit":"psi","seq":4}
    EOF

- Install busy box
    sudo apt update
    sudo apt install busybox-static

- Create initramfs folder
    cd ~/qemu-lab
    mkdir -p initramfs/bin initramfs/proc initramfs/sys initramfs/dev logs scripts

- Copy busy box to initramfs
    cp /scripts/init initramfs

    chmod +x initramfs/init

→ package it as telemetry-initramfs.cpio.gz
    cd ~/qemu-lab/initramfs
    find . | cpio -H newc -o | gzip > ../telemetry-initramfs.cpio.gz
    cd ..
- Find your Ubuntu kernel
    ls /boot/vmlinuz-*

- Create a local readable copy of Linux kernel
    sudo cp /boot/vmlinuz-6.17.0-23-generic ~/qemu-lab/kernel/
    sudo chown $USER:$USER ~/qemu-lab/kernel/vmlinuz-6.17.0-23-generic
